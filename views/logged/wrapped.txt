  <script src="https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.2/gsap.min.js"></script>
  <script src="https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.2/ScrollTrigger.min.js"></script>

  <script>
    gsap.registerPlugin(ScrollTrigger);

    // 1. DrawSVG Line Animation
    gsap.to("#leaf-path", { strokeDashoffset: 0, duration: 2, ease: "power2.inOut" });

    // 2. Slide 2 Counter Animation
    let wasteStat = { val: 0 };
    gsap.to(wasteStat, {
      val: 4,
      duration: 1.5,
      ease: "power2.out",
      scrollTrigger: { trigger: "#slide-2", start: "top 60%" },
      onUpdate: () => document.getElementById("waste-num").innerText = Math.floor(wasteStat.val) + " kg"
    });

    // 3. Slide 3 Streak Counter Animation
    let streakStat = { val: 0 };
    gsap.to(streakStat, {
      val: 5,
      duration: 1.5,
      ease: "power2.out",
      scrollTrigger: { trigger: "#slide-3", start: "top 60%" },
      onUpdate: () => document.getElementById("streak-num").innerText = Math.floor(streakStat.val)
    });

    // 4. Slide Card Reveals
    gsap.utils.toArray(".card").forEach((card) => {
      gsap.from(card, {
        scale: 0.8,
        opacity: 0,
        duration: 1,
        ease: "back.out(1.7)",
        scrollTrigger: { trigger: card, start: "top 75%" }
      });
    });

    // 5. Mouse Trail Effect
    const canvas = document.getElementById("trail-canvas");
    const ctx = canvas.getContext("2d");
    let points = [];

    function resizeCanvas() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
    window.addEventListener("resize", resizeCanvas);
    resizeCanvas();

    window.addEventListener("mousemove", (e) => {
      points.push({ x: e.clientX, y: e.clientY, alpha: 1 });
    });

    function drawTrail() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      for (let i = 0; i < points.length; i++) {
        let p = points[i];
        ctx.beginPath();
        ctx.arc(p.x, p.y, 4, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(224, 163, 59, ${p.alpha})`;
        ctx.fill();
        p.alpha -= 0.02;
      }
      points = points.filter(p => p.alpha > 0);
      requestAnimationFrame(drawTrail);
    }
    drawTrail();

    // 6. Export Card to PNG Function
    function exportStory() {
      const element = document.getElementById('export-card');
      html2canvas(element, { backgroundColor: '#16231C', scale: 2 }).then(canvas => {
        const link = document.createElement('a');
        link.href = canvas.toDataURL("image/png");
        link.download = 'eco-wrapped-2026.png';
        link.click();
      });
    }
  </script>
